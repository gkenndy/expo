import * as React from 'react';
import { useState } from 'react';
import { SafeAreaView, View, Text, TextInput, Pressable, FlatList, ScrollView, StyleSheet } from 'react-native';

function Button({ title, onPress, variant='primary' }) {
  return (
    <Pressable onPress={onPress} style={[styles.btn, variant==='secondary' && styles.btnSecondary]}>
      <Text style={[styles.btnText, variant==='secondary' && styles.btnTextSecondary]}>{title}</Text>
    </Pressable>
  );
}

function Tag({ text, color='#0ea5e9' }) {
  return <Text style={[styles.tag, {borderColor: color, color}]}>{text}</Text>;
}

const currency = (n)=> `R$ ${n.toFixed(2).replace('.',',')}`;
const gerarNumeroAIT = () => `AIT-ITA-${new Date().getFullYear()}-${Math.floor(Math.random()*900000+100000)}`;

const seed = [
  { id:'1', numero:gerarNumeroAIT(), placa:'ABC1D23', descricao:'Estacionar em local proibido', valor:195.23, status:'AUTUAÇÃO', quando:new Date() },
  { id:'2', numero:gerarNumeroAIT(), placa:'XYZ9E88', descricao:'Avançar sinal vermelho', valor:293.47, status:'DEFESA', quando:new Date() },
];

function AppAgente({ onEmitido }) {
  const [step, setStep] = useState(1);
  const [placa, setPlaca] = useState('');
  const [renavam, setRenavam] = useState('');
  const [cnh, setCnh] = useState('');
  const [desc, setDesc] = useState('Estacionar em local proibido');
  const [valor, setValor] = useState('195.23');
  const [gps, setGps] = useState(null);
  const [evid, setEvid] = useState(0);

  return (
    <View style={{gap:12}}>
      <Text style={styles.h2}>APP Agente — Emissão de AIT</Text>
      {step===1 && (
        <View style={{gap:8}}>
          <Text>Placa</Text>
          <TextInput style={styles.input} placeholder="ABC1D23" value={placa} onChangeText={(t)=>setPlaca(t.toUpperCase())}/>
          <Text>RENAVAM</Text>
          <TextInput style={styles.input} placeholder="00000000000" value={renavam} onChangeText={setRenavam}/>
          <Text>CNH (se identificado)</Text>
          <TextInput style={styles.input} placeholder="00000000000" value={cnh} onChangeText={setCnh}/>
          <Text>Descrição</Text>
          <TextInput style={[styles.input, {height:80}]} multiline value={desc} onChangeText={setDesc}/>
          <Text>Valor</Text>
          <TextInput style={styles.input} keyboardType="decimal-pad" value={valor} onChangeText={setValor}/>
          <View style={styles.rowBetween}>
            <Text style={{color:'#64748b'}}>{gps ? `GPS ok` : 'Sem localização'}</Text>
            <Button title="Capturar GPS" variant="secondary" onPress={()=>setGps({lat:-4.276, lng:-55.992})}/>
          </View>
          <Button title="Próximo" onPress={()=>setStep(2)}/>
        </View>
      )}
      {step===2 && (
        <View style={{gap:8}}>
          <View style={styles.rowGap}>
            <Button title="Adicionar Foto" variant="secondary" onPress={()=>setEvid(e=>e+1)}/>
            <Button title="Adicionar Vídeo" variant="secondary" onPress={()=>setEvid(e=>e+1)}/>
          </View>
          <Text style={{color:'#64748b'}}>Evidências: {evid}</Text>
          <View style={styles.rowBetween}>
            <Button title="Voltar" variant="secondary" onPress={()=>setStep(1)}/>
            <Button title="Próximo" onPress={()=>setStep(3)}/>
          </View>
        </View>
      )}
      {step===3 && (
        <View style={{gap:8}}>
          <Text style={styles.h3}>Revisão</Text>
          <Text>Placa: <Text style={styles.bold}>{placa || '—'}</Text></Text>
          <Text>RENAVAM: <Text style={styles.bold}>{renavam || '—'}</Text></Text>
          <Text>CNH: <Text style={styles.bold}>{cnh || '—'}</Text></Text>
          <Text>Local: <Text style={styles.bold}>{gps ? 'Av. Belém, 123 - Centro' : '—'}</Text></Text>
          <Text>Descrição: <Text style={styles.bold}>{desc}</Text></Text>
          <Text>Valor: <Text style={styles.bold}>{currency(Number(valor||'0'))}</Text></Text>
          <Text>Evidências: <Text style={styles.bold}>{evid}</Text></Text>
          <View style={styles.rowBetween}>
            <Button title="Voltar" variant="secondary" onPress={()=>setStep(2)}/>
            <Button title="Emitir AIT" onPress={()=>{
              onEmitido({
                id: String(Math.random()),
                numero: gerarNumeroAIT(),
                placa: placa||'SEM-PLACA',
                descricao: desc,
                valor: Number(valor||'0'),
                status: 'AUTUAÇÃO',
                quando: new Date()
              });
              setStep(1); setPlaca(''); setRenavam(''); setCnh(''); setDesc('Estacionar em local proibido'); setValor('195.23'); setGps(null); setEvid(0);
            }}/>
          </View>
        </View>
      )}
    </View>
  );
}

function Backoffice({ aits, onStatus }) {
  return (
    <View style={{gap:12}}>
      <Text style={styles.h2}>Backoffice — Autos de Infração</Text>
      <FlatList
        data={aits}
        keyExtractor={(i)=>i.id}
        ItemSeparatorComponent={()=> <View style={{height:8}}/>}
        renderItem={({item})=>(
          <View style={styles.card}>
            <Text style={styles.bold}>{item.numero}</Text>
            <Text>{item.placa} — {item.descricao}</Text>
            <View style={styles.rowBetween}>
              <Tag text={item.status}/>
              <View style={styles.rowGap}>
                <Button title="Emitir NA" variant="secondary" onPress={()=>onStatus(item.id, 'PENALIDADE')}/>
                <Button title="Julg. Defesa" variant="secondary" onPress={()=>onStatus(item.id, 'PENALIDADE')}/>
              </View>
            </View>
          </View>
        )}
      />
    </View>
  );
}

function Portal({ aits }) {
  const [q, setQ] = useState('');
  const results = aits.filter(a => a.placa.toLowerCase().includes(q.toLowerCase()) || a.numero.toLowerCase().includes(q.toLowerCase()));
  const [selected, setSelected] = useState(null);

  return (
    <View style={{gap:12}}>
      <Text style={styles.h2}>Portal do Cidadão</Text>
      <TextInput style={styles.input} placeholder="ABC1D23 ou AIT-ITA-..." value={q} onChangeText={setQ}/>
      <ScrollView style={{maxHeight: 220}}>
        {results.map((a)=>(
          <Pressable key={a.id} onPress={()=>setSelected(a)} style={[styles.card, selected?.id===a.id && {borderColor:'#0f766e'}]}>
            <Text style={styles.bold}>{a.numero}</Text>
            <Text style={{color:'#64748b'}}>{a.placa} — {a.quando.toLocaleDateString()}</Text>
          </Pressable>
        ))}
        {results.length===0 && <Text style={{color:'#64748b'}}>Nenhum resultado</Text>}
      </ScrollView>
      {selected && (
        <View style={styles.card}>
          <Text style={styles.bold}>{selected.numero} — {selected.placa}</Text>
          <Text>{selected.descricao}</Text>
          <Text>Status: <Text style={styles.bold}>{selected.status}</Text></Text>
          <Text>Valor: <Text style={styles.bold}>{currency(selected.valor)}</Text></Text>
          <View style={styles.rowGap}>
            <Button title="PIX (exemplo)" variant="secondary" onPress={()=>{}}/>
            <Button title="Boleto (exemplo)" variant="secondary" onPress={()=>{}}/>
          </View>
        </View>
      )}
    </View>
  );
}

export default function App() {
  const [tab, setTab] = useState('agente');
  const [aits, setAits] = useState(seed);

  const handleEmitido = (ait) => setAits(prev => [ait, ...prev]);
  const handleStatus = (id, novo) => setAits(prev => prev.map(a => a.id===id ? {...a, status: novo} : a));

  return (
    <SafeAreaView style={{flex:1}}>
      <View style={styles.topbar}>
        <Text style={styles.topbarTitle}>COTRAN — Itaituba</Text>
        <Text style={styles.topbarSub}>Protótipo • Multas</Text>
      </View>

      <View style={{padding:16, gap:16, flex:1}}>
        <View style={styles.tabs}>
          <Pressable onPress={()=>setTab('agente')} style={[styles.tabBtn, tab==='agente' && styles.tabBtnActive]}><Text style={[styles.tabText, tab==='agente'&&styles.tabTextActive]}>APP Agente</Text></Pressable>
          <Pressable onPress={()=>setTab('backoffice')} style={[styles.tabBtn, tab==='backoffice' && styles.tabBtnActive]}><Text style={[styles.tabText, tab==='backoffice'&&styles.tabTextActive]}>Backoffice</Text></Pressable>
          <Pressable onPress={()=>setTab('portal')} style={[styles.tabBtn, tab==='portal' && styles.tabBtnActive]}><Text style={[styles.tabText, tab==='portal'&&styles.tabTextActive]}>Portal</Text></Pressable>
        </View>

        <ScrollView contentContainerStyle={{paddingBottom:24}}>
          {tab==='agente' && <AppAgente onEmitido={handleEmitido} />}
          {tab==='backoffice' && <Backoffice aits={aits} onStatus={handleStatus} />}
          {tab==='portal' && <Portal aits={aits} />}
        </ScrollView>
      </View>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  topbar:{backgroundColor:'#0f766e', padding:16},
  topbarTitle:{color:'#fff', fontSize:18, fontWeight:'700'},
  topbarSub:{color:'#d1fae5', fontSize:12},
  h2:{fontSize:18, fontWeight:'700'},
  h3:{fontSize:16, fontWeight:'700'},
  input:{borderWidth:1, borderColor:'#cbd5e1', borderRadius:8, padding:10, backgroundColor:'#fff'},
  btn:{backgroundColor:'#0ea5e9', paddingVertical:10, paddingHorizontal:14, borderRadius:10, alignSelf:'flex-start'},
  btnText:{color:'#fff', fontWeight:'700'},
  btnSecondary:{backgroundColor:'#fff', borderWidth:1, borderColor:'#0ea5e9'},
  btnTextSecondary:{color:'#0ea5e9'},
  rowBetween:{flexDirection:'row', alignItems:'center', justifyContent:'space-between'},
  rowGap:{flexDirection:'row', gap:8},
  tag:{borderWidth:1, paddingHorizontal:8, paddingVertical:4, borderRadius:999, fontSize:12},
  bold:{fontWeight:'700'},
  card:{borderWidth:1, borderColor:'#cbd5e1', borderRadius:12, padding:12, backgroundColor:'#fff', marginBottom:8},
  tabs:{flexDirection:'row', backgroundColor:'#e2e8f0', borderRadius:10, overflow:'hidden'},
  tabBtn:{flex:1, padding:10, alignItems:'center'},
  tabBtnActive:{backgroundColor:'#0f766e'},
  tabText:{color:'#334155', fontWeight:'700'},
  tabTextActive:{color:'#fff'}
});
